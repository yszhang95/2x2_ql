  def readout(current, thres=5, downtime=12, hold=16, reset=1):
      q = []
      tt = []
      Q = np.add.accumulate(current)
      tmin = 0  # earliest allowed trigger
      while np.any(Q > thres):
          it = np.argmax(Q > thres)
          if Q[it + hold] > thres:
              # triggered
              # reset charge
              Q[it + hold + reset + 1 :] = (
                  Q[it + hold + reset + 1 :] - Q[it + hold + reset]
              )
              q.append(Q[it + hold])
              tt.append((np.max([0, tmin-downtime+reset]), it, it + hold + 1))
          tmin = it + hold + downtime + 1
          Q[:tmin] = 0
      return np.array(q), np.array(tt)

  def build_block_ops(N, K):
      assert N % K == 0, "N must be divisible by K"
      B = N // K
      oneB = np.ones((B, 1))
      I_K = np.eye(K)

      R = np.kron(I_K, oneB.T)          # K x N
      C = np.kron(I_K, oneB)            # N x K

      Wb = np.zeros((B, B-1))
      for j in range(B - 1):
          Wb[j, j] = 1.0
          Wb[B - 1, j] = -1.0
      W = np.kron(I_K, Wb)              # N x (N-K)

      assert np.allclose(R @ C, B * np.eye(K))
      assert np.allclose(R @ W, 0)

      return B, R, C, W

  def solve_block_ridge_full(A, y, C, W, lam):
      CtAt = C.T @ A.T
      WtAt = W.T @ A.T

      # print(W.T@W)

      A11 = CtAt @ A @ C
      A12 = CtAt @ A @ W
      A21 = A12.T
      # A22 = WtAt @ A @ W + lam * np.eye(W.shape[1])
      A22 = WtAt @ A @ W + lam * (W.T@W)

      rhs1 = CtAt @ y
      rhs2 = WtAt @ y
      KKT = np.block([[A11, A12],
                      [A21, A22]])

      rhs = np.vstack([rhs1, rhs2])

      # sol = np.linalg.solve(KKT, rhs)
      sol = np.linalg.lstsq(KKT, rhs)[0]
      # sol = scipy.optimize.nnls(KKT, rhs)[0]
      u = sol[:C.shape[1]]
      v = sol[C.shape[1]:]
      return u, v

  def baseline_identifiable_x(A, R, y):
      A_plus = np.linalg.pinv(A)
      return R @ A_plus @ y
